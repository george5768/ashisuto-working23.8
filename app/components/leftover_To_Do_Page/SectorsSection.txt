  { name: 'Construction', icon: '/images/construction.jpg', description: 'Smarter project management with connected analytics and IoT data.' },
  { name: 'Manufacturing', icon: '/images/manufacturing.jpg', description: 'Smart factory automation and production line intelligence.' },
  { name: 'Semiconductor', icon: '/images/semiconductor.jpg', description: 'Precision process optimization and supply chain intelligence.' },
  { name: 'Healthcare', icon: '/images/healthcare.jpg', description: 'Secure patient data solutions and operational efficiency.' },
  { name: 'Public Sector', icon: '/images/public-sector.jpg', description: 'Modernizing government services for greater citizen impact.' },
];

export default function SectorsSection() {
  return (
    <section className="bg-gray-950 py-24 px-4 sm:px-6 lg:px-8">
      <div className="max-w-7xl mx-auto">

        {/* Header */}
        <motion.div
          initial={{ opacity: 0, y: 24 }}
          whileInView={{ opacity: 1, y: 0 }}
          viewport={{ once: true, amount: 0.3 }}
          transition={{ duration: 0.6 }}
          className="text-center mb-14"
        >
          <span className="inline-flex items-center gap-2 px-4 py-1.5 rounded-full text-xs font-bold uppercase tracking-widest text-orange-400 bg-orange-400/10 border border-orange-400/20 mb-4">
            <span className="w-1.5 h-1.5 rounded-full bg-orange-400" />
            Industries We Serve
          </span>
          <h2 className="text-4xl sm:text-5xl font-extrabold text-white mb-4">
            Tailored Solutions for{' '}
            <span className="text-orange-400">Every Sector</span>
          </h2>
          <p className="text-gray-400 text-lg max-w-2xl mx-auto">
            Our AI platform adapts to the unique challenges of every industry — delivering measurable outcomes across sectors.
          </p>
        </motion.div>

        {/* Grid */}
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5">
          {sectors.map((sector, idx) => (
            <motion.div
              key={sector.name}
              initial={{ opacity: 0, y: 24 }}
              whileInView={{ opacity: 1, y: 0 }}
              viewport={{ once: true, amount: 0.15 }}
              transition={{ duration: 0.55, delay: idx * 0.08 }}
              className="group relative rounded-2xl overflow-hidden border border-white/5 hover:border-orange-500/40 transition-all duration-400 cursor-pointer"
            >
              {/* Background image */}
              <div className="relative h-52">
                <Image
                  src={sector.icon}
                  alt={sector.name}
                  fill
                  className="object-cover transition-transform duration-700 group-hover:scale-110"
                />
                <div className="absolute inset-0 bg-gradient-to-t from-gray-950 via-gray-950/60 to-gray-950/10 group-hover:via-gray-950/40 transition-all duration-400" />
              </div>
              {/* Text */}
              <div className="absolute bottom-0 left-0 right-0 p-5">
                <h3 className="text-lg font-bold text-white mb-1 group-hover:text-orange-400 transition-colors duration-300">
                  {sector.name}
                </h3>
                <p className="text-sm text-gray-400 leading-relaxed opacity-0 group-hover:opacity-100 translate-y-2 group-hover:translate-y-0 transition-all duration-300">
                  {sector.description}
                </p>
              </div>
              {/* Orange corner accent */}
              <div className="absolute top-4 right-4 w-2 h-2 rounded-full bg-orange-500 opacity-0 group-hover:opacity-100 transition-opacity duration-300" />
            </motion.div>
          ))}
        </div>
      </div>
    </section>
  );
}
